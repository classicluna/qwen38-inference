+def _strip_inline_comment(value):
+    """Drop the inline comment portion of a raw value.
+
+    A comment begins at an unquoted '#' or ';' that is preceded by
+    whitespace. While a run of double quotes is open, those characters are
+    literal and no comment is recognized.
+    """
+    result, i, n = [], 0, len(value)
+    in_quotes = False
+    while i < n:
+        ch = value[i]
+        if in_quotes and ch == '"':
+            # close the quoting run; keep it for the next strip
+            in_quotes = False
+            result.append(ch)
+        elif in_quotes:
+            result.append(ch)
+        elif ch == '"':
+            # enter quoted region
+            in_quotes = True
+            result.append(ch)
+        elif ch in "#;" and (i == 0 or value[i - 1].isspace()):
+            # unquoted comment marker following whitespace -> cut
+            break
+        else:
+            result.append(ch)
+        i += 1
+    return "".join(result).strip()
